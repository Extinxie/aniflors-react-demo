import { Elysia } from "elysia";
import { PrismaClient } from "@prisma/client";
import { fetchAnime } from "./anilist";

const prisma = new PrismaClient();

const app = new Elysia()
  // простая ошибка‑обёртка
  .error("ANIME_NOT_FOUND", "Anime not found in DB")
  // импортировать один тайтл по anilistId
  .post("/anime/import", async ({ body }) => {
    const { anilistId } = body as { anilistId: number };
    // 1️⃣ запрос к Anilist
    const data = await fetchAnime(anilistId);

    // 2️⃣ сохранить/обновить в Prisma
    const anime = await prisma.anime.upsert({
      where: { anilistId: data.anilistId },
      update: data,
      create: data
    });

    return anime;
  })
  // получить запись
  .get("/anime/:id", async ({ params }) => {
    const anime = await prisma.anime.findUnique({
      where: { id: Number(params.id) }
    });
    if (!anime) throw app.error("ANIME_NOT_FOUND");
    return anime;
  })
  // пример связывания с Shikimori (запрос к их API)
  .post("/anime/:id/link-shikimori", async ({ params, body }) => {
    const { shikimoriId } = body as { shikimoriId: number };
    return await prisma.anime.update({
      where: { id: Number(params.id) },
      data: { shikimoriId }
    });
  })
  // старт
  .listen(3000);

console.log(`🚀 Elysia listening on http://localhost:${app.server?.port}`);
